# OSTree boot logic with u-boot bootcount rollback.
#
# Adapted from meta-updater-raspberrypi, which targets a newer Yocto than this
# layer can use. One deliberate change from upstream, see bootcmd_load_f.
#
# Two uEnv.txt files are involved, do not confuse them:
#   this one            vfat partition 1, static, shipped in the image
#   /boot/loader/uEnv.txt  ext4 partition 2, REWRITTEN BY OSTREE on every
#                          deploy; carries kernel_image/bootargs for the current
#                          deployment and *2 variants for the previous one
bootlimit=3
devtype=mmc
devnum=0

# Normal boot. On the rollback path, altbootcmd runs instead.
bootcmd=if test "${rollback}" = "1"; then run altbootcmd; else run bootcmd_create_envfile; run bootcmd_otenv; run bootostree; if test ! "${upgrade_available}" = "1"; then setenv upgrade_available 1; saveenv; fi; reset; fi

bootcmd_resetvars=setenv kernel_image; setenv bootargs; setenv kernel_image2; setenv bootargs2

# Read what OSTree wrote for the current (and previous) deployment.
bootcmd_otenv=run bootcmd_resetvars; ext4load ${devtype} ${devnum}:2 ${scriptaddr} /boot/loader/uEnv.txt; env import -t ${scriptaddr} ${filesize} kernel_image bootargs kernel_image2 bootargs2

# CHANGED FROM UPSTREAM: upstream prefixes "/boot" here, because older OSTree
# wrote kernel_image relative to /boot. The OSTree in this build writes an
# absolute path already ("/boot/ostree/<os>-<csum>/vmlinuz-<ver>"), so adding
# the prefix would ask for /boot/boot/... and fail. Verify after an OSTree
# version bump: `cat /boot/loader/uEnv.txt` on the device.
bootcmd_load_f=ext4load ${devtype} ${devnum}:2 ${ramdisk_addr_r} ${kernel_image}

# kernel_image is a FIT image holding kernel-1 and ramdisk-1. The third
# argument is the firmware's device tree, which already carries the runtime
# fixups for this board (dma-ranges for the C0 stepping, and so on) -- prefer
# it over the FDT inside the FIT.
bootcmd_run=bootm ${ramdisk_addr_r}:kernel-1 ${ramdisk_addr_r}:ramdisk-1 ${fdt_addr}

bootostree=run bootcmd_load_f; run bootcmd_run

# Rollback: boot the previous deployment instead, and record that we did.
bootcmd_rollbackenv=setenv kernel_image ${kernel_image2}; setenv bootargs ${bootargs2}
bootcmd_set_rollback=if test ! "${rollback}" = "1"; then setenv rollback 1; setenv upgrade_available 0; saveenv; fi
altbootcmd=run bootcmd_otenv; run bootcmd_set_rollback; if test -n "${kernel_image2}"; then run bootcmd_rollbackenv; fi; run bootostree; reset

# bootcount lives in the saved environment, so the file has to exist.
bootcmd_create_envfile=if test ! -e ${devtype} ${devnum}:1 uboot.env; then saveenv; fi
